function inr(n) {
  return "₹" + Number(n).toLocaleString("en-IN");
}

function uid(prefix) {
  return prefix + Math.random().toString(36).slice(2, 8);
}

function timeAgo(ts) {
  const s = Math.floor((Date.now() - ts) / 1000);
  if (s < 60) return s + "s ago";
  const m = Math.floor(s / 60);
  if (m < 60) return m + "m ago";
  const h = Math.floor(m / 60);
  return h + "h " + (m % 60) + "m ago";
}

/* ---------------- storage helpers ---------------- */

async function loadShared(key, fallback) {
  try {
    const res = await window.storage.get(key, true);
    return res ? JSON.parse(res.value) : fallback;
  } catch (e) {
    return fallback;
  }
}
async function saveShared(key, value) {
  try {
    await window.storage.set(key, JSON.stringify(value), true);
  } catch (e) {
    console.error("storage set failed", key, e);
  }
}

function TornEdge({ flip }) {
  const teeth = 18;
  return (
    <svg
      viewBox={`0 0 ${teeth * 10} 10`}
      preserveAspectRatio="none"
      style={{ width: "100%", height: 10, display: "block", transform: flip ? "rotate(180deg)" : "none" }}
    >
      {Array.from({ length: teeth }).map((_, i) => (
        <circle key={i} cx={i * 10 + 5} cy={flip ? 10 : 0} r="5" fill={COLORS.paper} />
      ))}
    </svg>
  );
}

function Stepper({ qty, onChange }) {
  return (
    <div style={{ display: "flex", alignItems: "center", gap: 10 }}>
      <button
        onClick={() => onChange(Math.max(0, qty - 1))}
        style={stepBtnStyle}
        aria-label="Decrease quantity"
      >
        −
      </button>
      <span style={{ fontFamily: "'IBM Plex Mono', monospace", fontWeight: 600, minWidth: 16, textAlign: "center" }}>{qty}</span>
      <button onClick={() => onChange(qty + 1)} style={stepBtnStyle} aria-label="Increase quantity">
        +
      </button>
    </div>
  );
}

function CustomerView({ menu, orders, placeOrder, table, setTable }) {
  const [category, setCategory] = useState(CATEGORIES[0]);
  const [cart, setCart] = useState({}); // itemId -> qty
  const [cartOpen, setCartOpen] = useState(false);
  const [notes, setNotes] = useState("");
  const [payment, setPayment] = useState("cash");
  const [myOrderId, setMyOrderId] = useState(null);
  const [placing, setPlacing] = useState(false);

  const availableInCat = menu.filter((m) => m.category === category && m.available);
  const cartItems = Object.entries(cart).filter(([, q]) => q > 0);
  const cartCount = cartItems.reduce((s, [, q]) => s + q, 0);
  const cartTotal = cartItems.reduce((s, [id, q]) => {
    const item = menu.find((m) => m.id === id);
    return s + (item ? item.price * q : 0);
  }, 0);

  const setQty = (id, q) => setCart((c) => ({ ...c, [id]: q }));

  const myOrder = orders.find((o) => o.id === myOrderId);

  async function handlePlaceOrder() {
    if (cartItems.length === 0) return;
    setPlacing(true);
    const order = {
      id: uid("o"),
      table,
      items: cartItems.map(([id, qty]) => {
        const m = menu.find((mi) => mi.id === id);
        return { itemId: id, name: m.name, price: m.price, qty };
      }),
      notes,
      payment,
      status: "new",
      paid: false,
      createdAt: Date.now(),
    };
    await placeOrder(order);
    setMyOrderId(order.id);
    setCart({});
    setNotes("");
    setCartOpen(false);
    setPlacing(false);
  }

  if (myOrder) {
    const stepIdx = STATUS_FLOW.indexOf(myOrder.status);
    return (
      <div style={{ maxWidth: 480, margin: "0 auto", padding: "28px 16px 60px" }}>
        <Header table={table} />
        <div style={{ marginTop: 22 }}>
          <Ticket code={"#" + myOrder.id.slice(1, 4).toUpperCase()} stampColor={STATUS_COLOR[myOrder.status]}>
            <div style={{ fontFamily: "Fraunces, serif", fontSize: 22, color: COLORS.ink, marginBottom: 2 }}>
              Order placed
            </div>
            <div style={{ fontFamily: "'IBM Plex Mono', monospace", fontSize: 12, color: "#8A8375", marginBottom: 16 }}>
              Table {myOrder.table} · {timeAgo(myOrder.createdAt)}
            </div>

            <div style={{ display: "flex", justifyContent: "space-between", marginBottom: 14 }}>
              {STATUS_FLOW.map((s, i) => (
                <div key={s} style={{ textAlign: "center", flex: 1, opacity: i <= stepIdx ? 1 : 0.35 }}>
                  <div
                    style={{
                      width: 12,
                      height: 12,
                      borderRadius: "50%",
                      background: i <= stepIdx ? STATUS_COLOR[myOrder.status] : COLORS.line,
                      margin: "0 auto 6px",
                    }}
                  />
                  <div style={{ fontSize: 10, fontFamily: "'IBM Plex Mono', monospace", textTransform: "uppercase", letterSpacing: "0.05em" }}>
                    {STATUS_LABEL[s]}
                  </div>
                </div>
              ))}
            </div>

            <div style={{ borderTop: `1px dashed ${COLORS.line}`, paddingTop: 12 }}>
              {myOrder.items.map((it) => (
                <div key={it.itemId} style={{ display: "flex", justifyContent: "space-between", fontSize: 14, marginBottom: 4 }}>
                  <span>{it.qty}× {it.name}</span>

                </div>
              ))}

                <span>Total</span>
                <span style={{ fontFamily: "'IBM Plex Mono', monospace" }}>{inr(myOrder.items.reduce((s, it) => s + it.price * it.qty, 0))}</span>
              </div>

                Pay at counter · {myOrder.payment === "cash" ? "Cash" : "UPI"}
                {myOrder.paid && <span style={{ color: COLORS.sage, fontWeight: 600 }}> · Paid</span>}
              </div>
            </div>
          </Ticket>
          <button
            onClick={() => setMyOrderId(null)}
            style={{ ...primaryBtn, background: "transparent", color: COLORS.copper, border: `1px solid ${COLORS.copper}`, marginTop: 18, width: "100%" }}
          >
            Order more
          </button>
        </div>
      </div>
    );
  }

  return (
    <div style={{ maxWidth: 480, margin: "0 auto", paddingBottom: cartCount ? 90 : 30 }}>
      <div style={{ padding: "28px 16px 0" }}>
        <Header table={table} setTable={setTable} />
      </div>

      <div style={{ display: "flex", gap: 8, overflowX: "auto", padding: "18px 16px 10px", scrollbarWidth: "none" }}>
        {CATEGORIES.map((c) => (
          <button
            key={c}
            onClick={() => setCategory(c)}
            style={{
              whiteSpace: "nowrap",
              padding: "8px 14px",
              borderRadius: 999,
              border: `1px solid ${category === c ? COLORS.copper : COLORS.line}`,
              background: category === c ? COLORS.copper : "transparent",
              color: category === c ? "#fff" : COLORS.ink,
              fontFamily: "Inter, sans-serif",
              fontSize: 13,
              fontWeight: 600,
              cursor: "pointer",
            }}
          >
            {c}
          </button>
        ))}
      </div>

      <div style={{ padding: "6px 16px" }}>
        {availableInCat.length === 0 && (
          <div style={{ color: "#8A8375", fontSize: 14, padding: "24px 0", textAlign: "center" }}>Nothing here right now — try another category.</div>
        )}
        {availableInCat.map((item) => (
          <div
            key={item.id}
            style={{
              display: "flex",
              justifyContent: "space-between",
              alignItems: "center",
              padding: "14px 0",
              borderBottom: `1px solid ${COLORS.line}`,
              gap: 12,
            }}

            </div>

            <button onClick={handlePlaceOrder} disabled={placing} style={{ ...primaryBtn, width: "100%", marginTop: 16 }}>

            </button>
          </div>
        </div>
      )}
    </div>
  );
}

  return (

        {["overview", "menu", "orders"].map((t) => (
          <button
            key={t}

              background: "none",
              border: "none",
              padding: "10px 4px",
              marginRight: 18,
              fontFamily: "Inter, sans-serif",
              fontWeight: 600,
              fontSize: 14,
              textTransform: "capitalize",
              color: tab === t ? COLORS.ink : "#8A8375",
              borderBottom: tab === t ? `2px solid ${COLORS.copper}` : "2px solid transparent",
              cursor: "pointer",
            }}
          >
            {t}
          </button>
        ))}
      </div>

      {tab === "overview" && (

            <StatCard label="Revenue (paid)" value={inr(revenue)} />

function MenuEditor({ menu, setMenu }) {
  const [newItem, setNewItem] = useState({ name: "", price: "", category: CATEGORIES[0], desc: "", veg: true });

  function update(id, patch) {
    setMenu(menu.map((m) => (m.id === id ? { ...m, ...patch } : m)));
  }
  function remove(id) {
    setMenu(menu.filter((m) => m.id !== id));
  }
  function add() {
    if (!newItem.name || !newItem.price) return;
    setMenu([...menu, { id: uid("m"), name: newItem.name, desc: newItem.desc, price: Number(newItem.price), category: newItem.category, veg: newItem.veg, available: true }]);
    setNewItem({ name: "", price: "", category: newItem.category, desc: "", veg: true });
  }

  return (
    <div>
      {CATEGORIES.map((cat) => (
        <div key={cat} style={{ marginBottom: 22 }}>
          <div style={{ fontFamily: "Fraunces, serif", fontSize: 17, marginBottom: 8 }}>{cat}</div>
          {menu.filter((m) => m.category === cat).map((item) => (
            <div key={item.id} style={{ display: "flex", alignItems: "center", gap: 10, padding: "8px 0", borderBottom: `1px solid ${COLORS.line}` }}>
              <VegDot veg={item.veg} />
              <div style={{ flex: 1 }}>
                <div style={{ fontSize: 14, fontWeight: 600, opacity: item.available ? 1 : 0.4 }}>{item.name}</div>
                <div style={{ fontSize: 12, color: "#8A8375" }}>{item.desc}</div>
              </div>
              <input
                type="number"
                value={item.price}
                onChange={(e) => update(item.id, { price: Number(e.target.value) })}
                style={{ width: 70, padding: "5px 7px", border: `1px solid ${COLORS.line}`, borderRadius: 6, fontFamily: "'IBM Plex Mono', monospace", fontSize: 13 }}
              />
              <button
                onClick={() => update(item.id, { available: !item.available })}
                style={{
                  border: `1px solid ${item.available ? COLORS.sage : COLORS.line}`,
                  background: item.available ? COLORS.sage : "transparent",
                  color: item.available ? "#fff" : "#8A8375",
                  borderRadius: 8,
                  padding: "5px 10px",
                  fontSize: 12,
                  cursor: "pointer",
                  whiteSpace: "nowrap",
                }}
              >
                {item.available ? "Available" : "86'd"}
              </button>
              <button onClick={() => remove(item.id)} style={{ border: "none", background: "none", color: COLORS.rust, cursor: "pointer", fontSize: 13 }}>
                Remove
              </button>
            </div>
          ))}
        </div>
      ))}

      <div style={{ background: "#fff", border: `1px solid ${COLORS.line}`, borderRadius: 12, padding: 16, marginTop: 10 }}>
        <div style={{ fontFamily: "Fraunces, serif", fontSize: 16, marginBottom: 10 }}>Add menu item</div>
        <div style={{ display: "flex", flexWrap: "wrap", gap: 8 }}>
          <input placeholder="Name" value={newItem.name} onChange={(e) => setNewItem({ ...newItem, name: e.target.value })} style={inputStyle} />
          <input placeholder="Description" value={newItem.desc} onChange={(e) => setNewItem({ ...newItem, desc: e.target.value })} style={{ ...inputStyle, flex: 2 }} />
          <input placeholder="Price" type="number" value={newItem.price} onChange={(e) => setNewItem({ ...newItem, price: e.target.value })} style={{ ...inputStyle, width: 90 }} />
          <select value={newItem.category} onChange={(e) => setNewItem({ ...newItem, category: e.target.value })} style={inputStyle}>
            {CATEGORIES.map((c) => <option key={c} value={c}>{c}</option>)}
          </select>
          <select value={newItem.veg ? "veg" : "nonveg"} onChange={(e) => setNewItem({ ...newItem, veg: e.target.value === "veg" })} style={{ ...inputStyle, flex: "none", minWidth: 90 }}>
            <option value="veg">Veg</option>
            <option value="nonveg">Non-veg</option>
          </select>

        </div>
      </div>
    </div>
  );

function readUrlParams() {
  try {
    const p = new URLSearchParams(window.location.search);
    const role = p.get("role");
    const table = p.get("table");
    return {
      role: ["customer", "staff", "admin"].includes(role) ? role : null,
      table: table ? Number(table) : null,
    };
  } catch (e) {
    return { role: null, table: null };
  }
}

export default function App() {
  const urlParams = useRef(readUrlParams()).current;
  const [role, setRole] = useState(urlParams.role || "customer");
  const [table, setTable] = useState(urlParams.table || 4);
  const forcedRole = !!urlParams.role; // true when opened from a table/staff/admin QR link — hides the demo switcher
  const lockedTable = !!urlParams.table; // true when a specific table was passed in the link
  const [menu, setMenuState] = useState(DEFAULT_MENU);
  const [orders, setOrdersState] = useState([]);
  const [ready, setReady] = useState(false);
  const pollRef = useRef(null);

  useEffect(() => {
    (async () => {
      const [m, o] = await Promise.all([loadShared("cf-menu", DEFAULT_MENU), loadShared("cf-orders", [])]);
      setMenuState(m);
      setOrdersState(o);
      setReady(true);
    })();
  }, []);

  useEffect(() => {
    pollRef.current = setInterval(async () => {
      const o = await loadShared("cf-orders", []);
      setOrdersState(o);
    }, 3500);
    return () => clearInterval(pollRef.current);
  }, []);

  const setMenu = useCallback((next) => {
    setMenuState(next);
    saveShared("cf-menu", next);
  }, []);

  const placeOrder = useCallback(async (order) => {
    const latest = await loadShared("cf-orders", []);
    const next = [...latest, order];
    setOrdersState(next);
    await saveShared("cf-orders", next);
  }, []);

  const advanceStatus = useCallback(async (orderId) => {
    const latest = await loadShared("cf-orders", []);
    const next = latest.map((o) => {
      if (o.id !== orderId) return o;
      const idx = STATUS_FLOW.indexOf(o.status);
      return { ...o, status: STATUS_FLOW[Math.min(idx + 1, STATUS_FLOW.length - 1)] };
    });
    setOrdersState(next);
    await saveShared("cf-orders", next);
  }, []);

  const markPaid = useCallback(async (orderId, paid) => {
    const latest = await loadShared("cf-orders", []);
    const next = latest.map((o) => (o.id === orderId ? { ...o, paid } : o));
    setOrdersState(next);
    await saveShared("cf-orders", next);
  }, []);

  const clearOrders = useCallback(async () => {
    setOrdersState([]);
    await saveShared("cf-orders", []);
  }, []);

  if (!ready) {
    return (
      <div style={{ minHeight: "100vh", background: COLORS.paper, display: "flex", alignItems: "center", justifyContent: "center", fontFamily: "Inter, sans-serif", color: "#8A8375" }}>
        Loading…
      </div>
    );
  }

  return (
    <div style={{ minHeight: "100vh", background: COLORS.paper, color: COLORS.ink, fontFamily: "Inter, sans-serif" }}>
      <style>{FONTS}</style>
      {!forcedRole && (
      <div
        style={{
          position: "sticky",
          top: 0,
          zIndex: 10,
          background: COLORS.ink,
          color: COLORS.paper,
          display: "flex",
          justifyContent: "center",
          gap: 4,
          padding: "8px",
        }}
      >
        {[
          ["customer", "Customer (QR view)"],
          ["staff", "Kitchen"],
          ["admin", "Admin"],
        ].map(([key, label]) => (
          <button
            key={key}
            onClick={() => setRole(key)}
            style={{
              background: role === key ? COLORS.copper : "transparent",
              color: role === key ? "#fff" : COLORS.paper,
              border: "none",
              borderRadius: 8,
              padding: "7px 14px",
              fontSize: 12.5,
              fontWeight: 600,
              cursor: "pointer",
              fontFamily: "Inter, sans-serif",
            }}
          >
            {label}
          </button>
        ))}
      </div>
      )}

      {role === "customer" && <CustomerView menu={menu} orders={orders} placeOrder={placeOrder} table={table} setTable={lockedTable ? null : setTable} />}
      {role === "staff" && <StaffView orders={orders} advanceStatus={advanceStatus} />}
      {role === "admin" && <AdminView menu={menu} setMenu={setMenu} orders={orders} markPaid={markPaid} clearOrders={clearOrders} />}
    </div>
  );
}
